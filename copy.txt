    <script>
        // --- 1. THREE.JS SCENE SETUP ---
        const container = document.getElementById('canvas-container');
        const scene = new THREE.Scene();
        scene.background = new THREE.Color('#f0f0f5'); // Soft studio background

        const camera = new THREE.PerspectiveCamera(45, window.innerWidth / window.innerHeight, 0.1, 100);
        camera.position.set(0, 15, 30); // Look slightly down

        const renderer = new THREE.WebGLRenderer({ antialias: true });
        renderer.setSize(window.innerWidth, window.innerHeight);
        renderer.shadowMap.enabled = true;
        renderer.shadowMap.type = THREE.PCFSoftShadowMap; // Softer shadows
        container.appendChild(renderer.domElement);

        const controls = new THREE.OrbitControls(camera, renderer.domElement);
        controls.enableDamping = true;
        controls.dampingFactor = 0.05;
        controls.maxPolarAngle = Math.PI / 2 - 0.05; // Don't go below ground
        controls.minDistance = 15;
        controls.maxDistance = 50;

        // --- 2. LIGHTING ---
        const ambientLight = new THREE.AmbientLight(0xffffff, 0.6); // Soft ambient
        scene.add(ambientLight);

        const dirLight = new THREE.DirectionalLight(0xfff0e6, 0.8);
        dirLight.position.set(10, 20, 10);
        dirLight.castShadow = true;
        dirLight.shadow.mapSize.width = 2048;
        dirLight.shadow.mapSize.height = 2048;
        scene.add(dirLight);

        // --- 3. MATERIALS & TEXTURES ---
        function createStripeTexture() {
            const canvas = document.createElement('canvas');
            canvas.width = 256; canvas.height = 256;
            const ctx = canvas.getContext('2d');
            ctx.fillStyle = 'white'; ctx.fillRect(0, 0, 256, 256);
            ctx.fillStyle = '#ff4d6d'; // Pinkish-red stripe
            
            ctx.translate(128, 128);
            ctx.rotate(Math.PI / 4); // 45 degrees
            for(let i = -200; i < 200; i += 40) {
                ctx.fillRect(i, -200, 20, 400);
            }
            
            const texture = new THREE.CanvasTexture(canvas);
            texture.wrapS = THREE.RepeatWrapping;
            texture.wrapT = THREE.RepeatWrapping;
            texture.repeat.set(2, 1);
            return texture;
        }

        const matDarkChoc = new THREE.MeshStandardMaterial({ color: 0x3a2016, roughness: 0.95 });
        const matLightChoc = new THREE.MeshStandardMaterial({ color: 0x7c4935, roughness: 0.9 });
        const matPurpleIcing = new THREE.MeshStandardMaterial({ color: 0x9b59b6, roughness: 0.3, metalness: 0.1 }); 
        const matWhiteIcing = new THREE.MeshStandardMaterial({ color: 0xffffff, roughness: 0.5 });
        const matCandle = new THREE.MeshStandardMaterial({ map: createStripeTexture(), roughness: 0.7 });
        const matFlame = new THREE.MeshBasicMaterial({ color: 0xffdd44, transparent: true, opacity: 0.9 });
        
        // Fruit Materials
        const matStrawberry = new THREE.MeshStandardMaterial({ color: 0xd32f2f, roughness: 0.3, metalness: 0.1 });
        const matLeaves = new THREE.MeshStandardMaterial({ color: 0x4caf50, roughness: 0.7 });
        const matBlueberry = new THREE.MeshStandardMaterial({ color: 0x283593, roughness: 0.2, metalness: 0.2 });
        
        // --- 4. BUILD THE CAKE ---
        const cakeGroup = new THREE.Group();

        // Floor / Plate
        const plateGeo = new THREE.CylinderGeometry(9.5, 9.5, 0.2, 64);
        const matPlate = new THREE.MeshStandardMaterial({ color: 0xffffff, roughness: 0.1 });
        const plate = new THREE.Mesh(plateGeo, matPlate);
        plate.position.y = -0.1;
        plate.receiveShadow = true;
        cakeGroup.add(plate);

        // Helper to create a cake tier
        function createTier(radius, height, yOffset) {
            const tierGroup = new THREE.Group();
            const segmentHeight = height / 3;

            // Dark Chocolate Bottom
            const geoDark = new THREE.CylinderGeometry(radius, radius, segmentHeight, 64);
            const meshDark = new THREE.Mesh(geoDark, matDarkChoc);
            meshDark.position.y = segmentHeight / 2;
            meshDark.castShadow = true; meshDark.receiveShadow = true;
            tierGroup.add(meshDark);

            // Light Chocolate Middle
            const geoLight = new THREE.CylinderGeometry(radius, radius, segmentHeight, 64);
            const meshLight = new THREE.Mesh(geoLight, matLightChoc);
            meshLight.position.y = segmentHeight * 1.5;
            meshLight.castShadow = true; meshLight.receiveShadow = true;
            tierGroup.add(meshLight);

            // Purple Icing Top
            const geoPink = new THREE.CylinderGeometry(radius, radius, segmentHeight, 64);
            const meshPink = new THREE.Mesh(geoPink, matPurpleIcing);
            meshPink.position.y = segmentHeight * 2.5;
            meshPink.castShadow = true; meshPink.receiveShadow = true;
            tierGroup.add(meshPink);

            // Smooth icing edge
            const edgeGeo = new THREE.TorusGeometry(radius, 0.3, 16, 64);
            const edge = new THREE.Mesh(edgeGeo, matPurpleIcing);
            edge.rotation.x = Math.PI / 2;
            edge.position.y = segmentHeight * 3 - 0.1;
            edge.castShadow = true;
            tierGroup.add(edge);

            // Drips
            const numDrips = Math.floor(radius * 3);
            for(let i=0; i<numDrips; i++) {
                const angle = (i / numDrips) * Math.PI * 2;
                const dripLength = 0.5 + Math.random() * 1.5;
                const dripGeo = new THREE.CylinderGeometry(0.3, 0.1, dripLength, 8);
                const drip = new THREE.Mesh(dripGeo, matPurpleIcing);
                drip.position.set(Math.cos(angle) * radius, segmentHeight * 3 - dripLength/2, Math.sin(angle) * radius);
                drip.castShadow = true;
                tierGroup.add(drip);
            }

            // --- WAVE PIPING DECORATIONS ---
            
            // Middle Layer Wave
            const wavePoints = [];
            for(let i=0; i<=100; i++) {
                const angle = (i / 100) * Math.PI * 2;
                const yWave = segmentHeight * 1.5 + Math.sin(angle * 8) * 0.4; // 8 waves around
                wavePoints.push(new THREE.Vector3(radius * Math.cos(angle), yWave, radius * Math.sin(angle)));
            }
            const waveCurve = new THREE.CatmullRomCurve3(wavePoints);
            waveCurve.closed = true;
            const waveGeo = new THREE.TubeGeometry(waveCurve, 100, 0.15, 8, true);
            const waveMesh = new THREE.Mesh(waveGeo, matWhiteIcing);
            waveMesh.castShadow = true;
            tierGroup.add(waveMesh);

            // Bottom Base Scalloped Wave
            const baseWavePoints = [];
            for(let i=0; i<=100; i++) {
                const angle = (i / 100) * Math.PI * 2;
                const rWave = radius + 0.15;
                const yBaseWave = 0.15 + Math.abs(Math.sin(angle * 12)) * 0.25; // Scallop effect
                baseWavePoints.push(new THREE.Vector3(rWave * Math.cos(angle), yBaseWave, rWave * Math.sin(angle)));
            }
            const baseWaveCurve = new THREE.CatmullRomCurve3(baseWavePoints);
            baseWaveCurve.closed = true;
            const baseWaveGeo = new THREE.TubeGeometry(baseWaveCurve, 100, 0.2, 16, true);
            const baseWaveMesh = new THREE.Mesh(baseWaveGeo, matWhiteIcing);
            baseWaveMesh.castShadow = true;
            tierGroup.add(baseWaveMesh);

            // --- FRUITS & SPRINKLES ---
            const topY = segmentHeight * 3;

            // Sprinkles
            const sprinkleColors = [0xffffff, 0xffeb3b, 0x03a9f4, 0x4caf50, 0xff9800];
            const sprinkleGeo = new THREE.CylinderGeometry(0.04, 0.04, 0.2, 5);
            for (let i = 0; i < 120; i++) {
                const sMat = new THREE.MeshStandardMaterial({ color: sprinkleColors[Math.floor(Math.random() * sprinkleColors.length)], roughness: 0.4 });
                const sprinkle = new THREE.Mesh(sprinkleGeo, sMat);
                const rSprinkle = Math.random() * (radius - 0.5);
                const angleSprinkle = Math.random() * Math.PI * 2;
                sprinkle.position.set(rSprinkle * Math.cos(angleSprinkle), topY + 0.05, rSprinkle * Math.sin(angleSprinkle));
                sprinkle.rotation.set(Math.random() * Math.PI, Math.random() * Math.PI, Math.random() * Math.PI);
                sprinkle.castShadow = true;
                tierGroup.add(sprinkle);
            }

            // Strawberries
            const strawGeo = new THREE.SphereGeometry(0.35, 16, 16);
            strawGeo.scale(1, 1.4, 1); // Make it conical
            const leafGeo = new THREE.ConeGeometry(0.12, 0.3, 3);
            
            for(let i=0; i<8; i++) {
                const angle = (i / 8) * Math.PI * 2 + (Math.PI / 8); // Offset to avoid candles
                const rStraw = radius - 0.8; 
                const strawGroup = new THREE.Group();
                
                const straw = new THREE.Mesh(strawGeo, matStrawberry);
                straw.castShadow = true;
                strawGroup.add(straw);

                // Add 4 tiny leaves on top
                for(let j=0; j<4; j++) {
                    const leaf = new THREE.Mesh(leafGeo, matLeaves);
                    leaf.position.y = 0.45;
                    leaf.rotation.x = Math.PI / 2;
                    leaf.rotation.z = (j / 4) * Math.PI * 2;
                    strawGroup.add(leaf);
                }

                strawGroup.position.set(rStraw * Math.cos(angle), topY + 0.4, rStraw * Math.sin(angle));
                strawGroup.rotation.x = -Math.PI / 6; // Tilt slightly outward
                strawGroup.rotation.y = angle;
                tierGroup.add(strawGroup);
            }

            // Blueberries
            const blueGeo = new THREE.SphereGeometry(0.18, 16, 16);
            for(let i=0; i<18; i++) {
                const angle = Math.random() * Math.PI * 2;
                const rBlue = 2.0 + Math.random() * (radius - 3.0); 
                const berry = new THREE.Mesh(blueGeo, matBlueberry);
                berry.position.set(rBlue * Math.cos(angle), topY + 0.18, rBlue * Math.sin(angle));
                berry.castShadow = true;
                tierGroup.add(berry);
            }

            tierGroup.position.y = yOffset;
            return tierGroup;
        }

        // Add Single Tier Cake
        const bottomTier = createTier(6, 4.5, 0);
        cakeGroup.add(bottomTier);
        
        // Add Candles
        const candles = [];
        const flames = [];
        const candleLights = [];
        const smokeParticles = [];

        function addCandle(x, z) {
            const candleGroup = new THREE.Group();
            
            // Wax
            const geoCandle = new THREE.CylinderGeometry(0.25, 0.25, 2.5, 16);
            const meshCandle = new THREE.Mesh(geoCandle, matCandle);
            meshCandle.position.y = 1.25;
            meshCandle.castShadow = true;
            candleGroup.add(meshCandle);

            // Wick
            const geoWick = new THREE.CylinderGeometry(0.05, 0.05, 0.4, 8);
            const matWick = new THREE.MeshStandardMaterial({color: 0x222222});
            const wick = new THREE.Mesh(geoWick, matWick);
            wick.position.y = 2.6;
            candleGroup.add(wick);

            // Flame
            const geoFlame = new THREE.ConeGeometry(0.2, 0.6, 16);
            geoFlame.translate(0, 0.3, 0); // shift origin to base
            const flame = new THREE.Mesh(geoFlame, matFlame);
            flame.position.y = 2.7;
            candleGroup.add(flame);
            flames.push(flame);

            // Flame Light (Creates the glowing effect on the cake)
            const light = new THREE.PointLight(0xffaa00, 1, 10);
            light.position.y = 3.0;
            candleGroup.add(light);
            candleLights.push(light);

            candleGroup.position.set(x, 4.5, z); // Set on top of the single tier
            cakeGroup.add(candleGroup);
            candles.push(candleGroup);
        }

        addCandle(-1.8, 0.5); // Left
        addCandle(0, -0.5);   // Middle
        addCandle(1.8, 0.5);  // Right

        scene.add(cakeGroup);

        // Move cake down slightly so it's centered
        cakeGroup.position.y = -2;

        // --- 5. ANIMATION & RENDER LOOP ---
        const clock = new THREE.Clock();
        let isBlownOut = false;

        function animate() {
            requestAnimationFrame(animate);
            const time = clock.getElapsedTime();

            // Flicker flames and lights
            if (!isBlownOut) {
                flames.forEach((flame, index) => {
                    const scaleModifier = 1 + Math.sin(time * 20 + index) * 0.1;
                    flame.scale.set(1, scaleModifier, 1);
                    // Slight sway
                    flame.rotation.z = Math.sin(time * 10 + index) * 0.1;
                    flame.rotation.x = Math.cos(time * 15 + index) * 0.1;
                    
                    // Match light intensity to flicker
                    candleLights[index].intensity = 0.8 + Math.sin(time * 20 + index) * 0.2;
                });
            } else {
                // Animate smoke particles
                for (let i = smokeParticles.length - 1; i >= 0; i--) {
                    let p = smokeParticles[i];
                    p.position.y += p.userData.vy;
                    p.position.x += p.userData.vx;
                    p.scale.multiplyScalar(1.02); // grow
                    p.material.opacity -= 0.01; // fade out
                    if (p.material.opacity <= 0) {
                        scene.remove(p);
                        smokeParticles.splice(i, 1);
                    }
                }
            }

            controls.update();
            renderer.render(scene, camera);
        }
        animate();

        // Handle Resize
        window.addEventListener('resize', () => {
            camera.aspect = window.innerWidth / window.innerHeight;
            camera.updateProjectionMatrix();
            renderer.setSize(window.innerWidth, window.innerHeight);
        });

        // --- 6. MICROPHONE LOGIC ---
        const startBtn = document.getElementById('start-btn');
        const manualBtn = document.getElementById('manual-btn');
        const resetBtn = document.getElementById('reset-btn');
        const btnGroup = document.getElementById('btn-group');
        const statusText = document.getElementById('status');
        const meterContainer = document.getElementById('meter-container');
        const meterFill = document.getElementById('meter');
        
        let audioContext;
        let analyser;
        let microphone;
        let isListening = false;
        let micAnimationId;
        
        const BLOW_THRESHOLD = 50;

        function spawnSmoke(x, y, z) {
            const smokeGeo = new THREE.SphereGeometry(0.2, 8, 8);
            const smokeMat = new THREE.MeshBasicMaterial({ color: 0x888888, transparent: true, opacity: 0.6 });
            
            for(let i=0; i<5; i++) {
                const particle = new THREE.Mesh(smokeGeo, smokeMat.clone());
                particle.position.set(x, y, z);
                particle.userData = {
                    vy: 0.02 + Math.random() * 0.03,
                    vx: (Math.random() - 0.5) * 0.02
                };
                scene.add(particle);
                smokeParticles.push(particle);
            }
        }

        function extinguishCandles() {
            if (isBlownOut) return;
            isBlownOut = true;

            // Hide 3D flames and lights, spawn smoke
            flames.forEach((flame, index) => {
                setTimeout(() => {
                    flame.visible = false;
                    candleLights[index].intensity = 0;
                    
                    // Get global position of the flame to spawn smoke
                    const worldPos = new THREE.Vector3();
                    flame.getWorldPosition(worldPos);
                    spawnSmoke(worldPos.x, worldPos.y, worldPos.z);
                    
                }, index * 100); // Stagger out
            });
            
            // UI Updates
            statusText.innerText = "Yay! Happy Birthday! 🎉";
            statusText.style.color = "#e91e63";
            isListening = false;
            
            if(microphone && microphone.mediaStream) {
                microphone.mediaStream.getTracks().forEach(track => track.stop());
            }
            if(audioContext) audioContext.close();
            if(micAnimationId) cancelAnimationFrame(micAnimationId);

            setTimeout(() => {
                btnGroup.classList.add('hidden');
                meterContainer.style.display = 'none';
                resetBtn.classList.remove('hidden');
            }, 1000);
        }

        resetBtn.addEventListener('click', () => {
            isBlownOut = false;
            flames.forEach((flame, index) => {
                flame.visible = true;
                candleLights[index].intensity = 1;
            });
            btnGroup.classList.remove('hidden');
            resetBtn.classList.add('hidden');
            statusText.innerText = "Ready!";
            statusText.style.color = "#555";
            meterContainer.style.display = 'none';
            meterFill.style.width = '0%';
        });

        manualBtn.addEventListener('click', extinguishCandles);

        startBtn.addEventListener('click', async () => {
            if (isListening) return;

            try {
                const stream = await navigator.mediaDevices.getUserMedia({ audio: true, video: false });
                
                audioContext = new (window.AudioContext || window.webkitAudioContext)();
                if (audioContext.state === 'suspended') {
                    await audioContext.resume();
                }
                
                analyser = audioContext.createAnalyser();
                microphone = audioContext.createMediaStreamSource(stream);
                microphone.connect(analyser);
                analyser.fftSize = 256;
                
                const bufferLength = analyser.frequencyBinCount;
                const dataArray = new Uint8Array(bufferLength);
                
                btnGroup.classList.add('hidden');
                statusText.innerText = "Listening... Blow into the mic! 🌬️";
                meterContainer.style.display = 'block';
                isListening = true;

                function detectBlow() {
                    if (!isListening) return;
                    analyser.getByteFrequencyData(dataArray);
                    
                    let sum = 0;
                    for (let i = 0; i < bufferLength; i++) { sum += dataArray[i]; }
                    let averageVolume = sum / bufferLength;

                    const volumePercent = Math.min((averageVolume / 100) * 100, 100);
                    meterFill.style.width = `${volumePercent}%`;
                    
                    if (averageVolume > BLOW_THRESHOLD) {
                        meterFill.style.backgroundColor = '#f44336';
                        extinguishCandles();
                    } else {
                        meterFill.style.backgroundColor = '#4CAF50';
                        micAnimationId = requestAnimationFrame(detectBlow);
                    }
                }
                detectBlow();

            } catch (err) {
                console.error("Microphone error:", err);
                statusText.innerText = "Mic access blocked. Use the Force Blow Out button!";
                statusText.style.color = "#d32f2f";
            }
        });
